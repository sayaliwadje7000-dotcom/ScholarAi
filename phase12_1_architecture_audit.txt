from collections.abc import Generator

from sqlalchemy import create_engine
from sqlalchemy.orm import Session, sessionmaker

from app.core.config import settings


if not settings.DATABASE_URL:
    raise RuntimeError(
        "DATABASE_URL is missing. Check your .env configuration."
    )


engine = create_engine(
    settings.DATABASE_URL,
    pool_pre_ping=True,
    pool_recycle=1800,
)


SessionLocal = sessionmaker(
    bind=engine,
    class_=Session,
    autoflush=False,
    autocommit=False,
    expire_on_commit=False,
)


def get_db() -> Generator[Session, None, None]:
    """
    Provides a database session to API endpoints.

    Automatically rolls back unfinished transactions on errors
    and closes the session after use.
    """
    db = SessionLocal()

    try:
        yield db

    except Exception:
        db.rollback()
        raise

    finally:
        db.close()

# ============================================================

class RecommendationBaseSchema(BaseModel):
    model_config = ConfigDict(
        extra="forbid",
        str_strip_whitespace=True,
    )


# ============================================================
# RECOMMENDATION GENERATION REQUEST
# ============================================================


class RecommendationRequest(RecommendationBaseSchema):
    """
    Public request schema for generating recommendations.

    User identity is deliberately excluded.
    The authenticated user must be obtained from
    the authentication dependency.
    """

    query: str | None = Field(
        default=None,
        min_length=1,
        max_length=1000,
        description="Optional topic or learning query.",
    )

    subject: str | None = Field(
        default=None,
        min_length=1,
        max_length=100,
    )

    resource_types: list[ResourceType] | None = Field(
        default=None,
        min_length=1,
        max_length=8,
    )

    difficulty_level: DifficultyLevel | None = None

    language: str | None = Field(
        default=None,
        min_length=2,
        max_length=30,
    )

    recommendation_type: RecommendationType = Field(
        default=RecommendationType.PERSONALIZED,
    )

    limit: int = Field(
        default=10,
        ge=1,
        le=50,
    )

    exclude_completed: bool = Field(
        default=True,
    )

    @field_validator("query", "subject", "language")
    @classmethod
    def normalize_optional_strings(
        cls,
        value: str | None,
    ) -> str | None:

        if value is None:
            return None

        value = value.strip()

        if not value:
            raise ValueError(
                "Value cannot contain only whitespace."
            )

        return value

    @field_validator("resource_types")
    @classmethod
    def validate_resource_types(
        cls,
        value: list[ResourceType] | None,
    ) -> list[ResourceType] | None:

        if value is None:
            return None

        if len(value) != len(set(value)):
            raise ValueError(
                "Duplicate resource types are not allowed."
            )

        return value

    @field_validator("language")
    @classmethod
    def normalize_language(
        cls,
        value: str | None,
    ) -> str | None:

        if value is None:
            return None

        return value.lower()

class InteractionSummary(RecommendationBaseSchema):
    """
    Internal summary of a user's resource interactions.
    """

    resource_id: UUID

    interaction_type: InteractionType

    rating: int | None = Field(
        default=None,
        ge=1,
        le=5,
    )

    duration_seconds: int | None = Field(
        default=None,
        ge=0,
    )

    occurred_at: datetime

    @model_validator(mode="after")
    def validate_rating(self):

        if self.interaction_type == InteractionType.RATE:
            if self.rating is None:
                raise ValueError(
                    "Rating is required for rate interactions."
                )

        elif self.rating is not None:
            raise ValueError(
                "Rating is only permitted for rate interactions."
            )

        return self


class LearningContext(RecommendationBaseSchema):
    """
    Internal representation of the authenticated user's
    learning preferences and interaction history.
    """

    user_id: UUID

    preferred_language: str | None = None

    education_level: str | None = None

    preferred_subjects: list[str] = Field(
        default_factory=list,
        max_length=50,
    )

    learning_goals: list[str] = Field(
        default_factory=list,
        max_length=50,
    )

    recent_interactions: list[InteractionSummary] = Field(
        default_factory=list,
        max_length=500,
    )

    @field_validator(
        "preferred_subjects",
        "learning_goals",
    )
    @classmethod
    def normalize_string_lists(
        cls,
        values: list[str],
    ) -> list[str]:

        normalized = []

        for value in values:

            value = value.strip()

            if not value:
                raise ValueError(
                    "List values cannot be empty."
                )

            normalized.append(value)

        return list(dict.fromkeys(normalized))

    """
    Normalized resource representation used internally
    by recommendation services.
    """

        min_length=1,
        max_length=255,

        max_length=100,

        if not value:

        cls,
        values: list[str],
    ) -> list[str]:

        normalized = [
            value.strip().lower()
            for value in values

        ]

        return list(dict.fromkeys(normalized))

class RecommendationWeights(RecommendationBaseSchema):
    """
    Configurable weights for the rule-based scoring engine.
    """

    topic_relevance: float = Field(
        default=0.35,
        ge=0,
        le=1,
    )

    learning_level: float = Field(
        default=0.20,
        ge=0,
        le=1,
    )

    prerequisite_compatibility: float = Field(
        default=0.15,
        ge=0,
        le=1,
    )

    resource_quality: float = Field(
        default=0.15,
        ge=0,
        le=1,
    )

    learning_objective: float = Field(
        default=0.10,
        ge=0,
        le=1,
    )

    freshness: float = Field(
        default=0.05,
        ge=0,
        le=1,
    )

    @model_validator(mode="after")
    def validate_weight_total(self):

        total = sum(
            (
                self.topic_relevance,
                self.learning_level,
                self.prerequisite_compatibility,
                self.resource_quality,
                self.learning_objective,
                self.freshness,
            )
        )

        if abs(total - 1.0) > 1e-9:
            raise ValueError(
                "Recommendation weights must sum to 1.0."
            )

        return self

class RecommendationListResponse(
    RecommendationBaseSchema
):
    """
    Paginated-style recommendation response.

    Pagination metadata is kept explicit so that
    cursor-based pagination can be introduced later.
    """

    recommendations: list[RecommendationResponse]

    total_results: int = Field(
        ge=0,
    )

    algorithm_version: str

    @model_validator(mode="after")
    def validate_result_count(self):

        if self.total_results != len(self.recommendations):
            raise ValueError(
                "total_results must equal the number "
                "of returned recommendations."
            )

        ranks = [
            recommendation.rank_position
            for recommendation in self.recommendations
        ]

        if len(ranks) != len(set(ranks)):
            raise ValueError(
                "Recommendation ranks must be unique."
            )

        return self

@router.get(
    "/",
    response_model=RecommendationHistoryAPIResponse,
)
def get_recommendations(
    limit: int = 20,
    offset: int = 0,
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    if not 1 <= limit <= 100:
        raise HTTPException(
            status_code=400,
            detail="Limit must be between 1 and 100.",
        )

    if offset < 0:
        raise HTTPException(
            status_code=400,
            detail="Offset cannot be negative.",
        )

    statement = (
        select(Recommendation)
        .where(
            Recommendation.user_id == current_user.id
        )
        .options(
            selectinload(Recommendation.resource)
        )
        .order_by(
            Recommendation.generated_at.desc(),
            Recommendation.rank_position.asc(),
        )
        .offset(offset)
        .limit(limit)
    )

    records = list(
        db.scalars(statement).all()
    )

    response_items = []

    for record in records:

        metadata = record.recommendation_metadata or {}

        explanation_data = metadata.get("explanation")

        if not explanation_data:
            continue

        explanation = RecommendationExplanation.model_validate(
            explanation_data
        )

        resource = record.resource

        response_items.append(
            RecommendationItemAPIResponse(
                recommendation_id=record.id,
                resource=RecommendedResourceAPIResponse(
                    id=resource.id,
                    title=resource.title,
                    description=resource.description,
                    resource_type=resource.resource_type,
                    url=resource.url,
                    provider=resource.provider,
                    subject=resource.subject,
                    language=resource.language,
                    difficulty_level=resource.difficulty_level,
                    estimated_duration_minutes=(
                        resource.estimated_duration_minutes
                    ),
                    quality_score=(
                        float(resource.quality_score)
                        if resource.quality_score is not None
                        else None
                    ),
                ),
                explanation=explanation,
                status=record.status,
                generated_at=record.generated_at,
            )
        )

    return RecommendationHistoryAPIResponse(
        total_results=len(response_items),
        recommendations=response_items,
    )


@router.post(
    "/{recommendation_id}/feedback",
    response_model=RecommendationFeedbackAPIResponse,
    status_code=status.HTTP_201_CREATED,
)
def submit_recommendation_feedback(
    recommendation_id: UUID,
    request: RecommendationFeedbackCreate,
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
):
    recommendation = db.scalar(
        select(Recommendation).where(
            Recommendation.id == recommendation_id,
            Recommendation.user_id == current_user.id,
        )
    )

    if recommendation is None:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="Recommendation not found.",
        )

    existing_feedback = db.scalar(
        select(RecommendationFeedback).where(
            RecommendationFeedback.recommendation_id
            == recommendation_id
        )
    )

    if existing_feedback is not None:
        raise HTTPException(
            status_code=status.HTTP_409_CONFLICT,
            detail="Feedback has already been submitted.",
        )

    feedback = RecommendationFeedback(
        recommendation_id=recommendation_id,
        feedback_type=request.feedback_type,
        comment=request.comment,
    )

    db.add(feedback)

    try:
        db.commit()
        db.refresh(feedback)

    except Exception as exc:
        db.rollback()

        raise HTTPException(
            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
            detail="Unable to save recommendation feedback.",
        ) from exc

    return RecommendationFeedbackAPIResponse(
        id=feedback.id,
        recommendation_id=feedback.recommendation_id,
        feedback_type=feedback.feedback_type,
        comment=feedback.comment,
        created_at=feedback.created_at,
    )

)

)

from app.services.recommendations.scoring_service import (
    RecommendationScoringService,
)

class RecommendationServiceError(Exception):
    pass


class RecommendationApplicationService:
    """
    Application-level orchestration for ScholarAI recommendations.

    Coordinates classification, context construction, scoring,
    filtering, ranking, explanations and persistence.
    """

    def __init__(
        self,
        classifier=None,
        context_builder=None,
        scoring_service=None,
        filtering_service=None,
        ranking_service=None,
        explanation_service=None,
    ):
        self.classifier = (
            classifier or ResourceClassifier()
        )

        self.context_builder = (
            context_builder or LearningContextBuilder()
        )

        self.scoring_service = (
            scoring_service or RecommendationScoringService()
        )

        self.filtering_service = (
            filtering_service or RecommendationFilteringService()
        )

        self.ranking_service = (
            ranking_service or RecommendationRankingService()
        )

        self.explanation_service = (
            explanation_service
            or RecommendationExplanationService()
        )

    def _load_interactions(
        self,
        db: Session,
        user_id: UUID,
    ):
        statement = (
            select(UserResourceInteraction)
            .where(
                UserResourceInteraction.user_id == user_id
            )
            .order_by(
                UserResourceInteraction.occurred_at.desc()
            )
            .limit(20)
        )

        return list(db.scalars(statement).all())

    def _load_resources(
        self,
        db: Session,
    ):
        statement = (
            select(StudyResource)
            .where(
                StudyResource.is_active.is_(True)
            )
            .options(
                selectinload(StudyResource.categories)
            )
            .order_by(
                StudyResource.created_at.desc()
            )
        )

        return list(db.scalars(statement).all())

    def generate(
        self,
        db: Session,
        user: User,
        request: RecommendationRequest,
    ):
        """
        Generate and persist personalized recommendations.

        All persistence occurs inside one transaction.
        """

        if not isinstance(user, User):
            raise TypeError("Expected authenticated User.")

        if not isinstance(request, RecommendationRequest):
            raise TypeError("Expected RecommendationRequest.")

        try:
            interactions = self._load_interactions(
                db,
                user.id,
            )

            resources = self._load_resources(db)

            context = self.context_builder.build(
                user=user,
                preferences=user.preferences,
                request=request,
                interactions=interactions,
            )

            classified_resources = [
                self.classifier.classify(resource)
                for resource in resources
            ]

            scored_candidates = []

            for resource, classification in zip(
                resources,
                classified_resources,
            ):
                scoring_result = self.scoring_service.score(
                    classification,
                    context,
                )

                if scoring_result is None:
                    continue

                scored_candidates.append(
                    (
                        classification,
                        scoring_result,
                    )
                )

            filtered_candidates = self.filtering_service.filter(
                candidates=scored_candidates,
                request=request,
                interactions=interactions,
            )

            ranked = self.ranking_service.rank(
                candidates=filtered_candidates,
                limit=request.limit,
            )

            explanations = self.explanation_service.explain_many(
                ranked_recommendations=ranked,
                context=context,
            )

            recommendation_records = []

            for ranked_item, explanation in zip(
                ranked,
                explanations,
            ):
                resource = ranked_item.resource

                database_resource = next(
                    (
                        item
                        for item in resources
                        if str(item.id) == str(resource.resource_id)
                    ),
                    None,
                )

                if database_resource is None:
                    continue

                record = Recommendation(
                    user_id=user.id,
                    resource_id=database_resource.id,
                    recommendation_type=request.recommendation_type.value,
                    score=ranked_item.score,
                    rank_position=ranked_item.rank_position,
                    reason=explanation.reason,
                    status="generated",
                    algorithm_version=ranked_item.algorithm_version,
                    recommendation_metadata={
                        "explanation": explanation.model_dump(mode="json"),
                        "explanation_version": explanation.explanation_version,
                        "context_version": context.context_version,
                    },
                )

                db.add(record)

                recommendation_records.append(
                    (record, database_resource, explanation)
                )

            db.flush()

            return {
                "request_id": uuid4(),
                "total_results": len(recommendation_records),
                "recommendations": recommendation_records,
                "generated_at": datetime.now(timezone.utc),
                "algorithm_version": self.scoring_service.ALGORITHM_VERSION,
            }

        except Exception:
            db.rollback()
            raise

from __future__ import annotations

from uuid import UUID

from sqlalchemy import select
from sqlalchemy.orm import Session

from app.models.user_resource_interaction import UserResourceInteraction

from app.schemas.content_recommendations import (
    ContentRecommendationRequest,
    ContentRecommendationResponse,
    ContentRecommendationItem,
)

from app.services.recommendations.content_based.query_context import (
    ContentQueryFilters,
)


class ContentRecommendationServiceError(Exception):
    """Base recommendation service error."""


class ContentBasedRecommendationService:

    ALGORITHM_VERSION = "content-based-v1"

    def __init__(
        self,
        query_context_service,
        similarity_service,
        feature_extractor,
        scoring_service,
        filtering_service,
        ranking_service,
    ):
        self.query_context_service = query_context_service
        self.similarity_service = similarity_service
        self.feature_extractor = feature_extractor
        self.scoring_service = scoring_service
        self.filtering_service = filtering_service
        self.ranking_service = ranking_service

    def _load_interactions(
        self,
        db: Session,
        user_id: UUID,
    ):
        """
        Load recent user interactions.

        A missing database session returns an empty interaction
        collection for isolated unit tests. Production API calls
        should always provide a valid database session.
        """

        if db is None:
            return []

        statement = (
            select(UserResourceInteraction)
            .where(
                UserResourceInteraction.user_id == user_id
            )
            .order_by(
                UserResourceInteraction.occurred_at.desc()
            )
            .limit(20)
        )

        return list(db.scalars(statement).all())

    def _build_context(
        self,
        db: Session,
        user_id: UUID,
        request: ContentRecommendationRequest,
    ):
        filters = ContentQueryFilters(
            subject=request.subject,
            language=request.language,
            difficulty_level=request.difficulty_level,
            resource_types=tuple(request.resource_types),
            exclude_completed=request.exclude_completed,
        )

        return self.query_context_service.build(
            db=db,
            user_id=user_id,
            query_text=request.query,
            filters=filters,
        )

    def _retrieve_candidates(
        self,
        db: Session,
        context,
        request: ContentRecommendationRequest,
    ):
        return self.similarity_service.find_similar_resources(
            db=db,
            query_vector=context.query_embedding,
            limit=min(request.limit * 5, 100),
            min_similarity=request.minimum_similarity,
            subject=request.subject,
            language=request.language,
        )

    def _score_candidates(
        self,
        similarity_results,
        context,
    ):
        candidates = []

        for result in similarity_results:
            resource = result.resource

            features = self.feature_extractor.extract(
                resource
            )

            if not self._has_usable_features(features):
                continue

            scoring_result = self.scoring_service.score(
                reference_features=context,
                candidate_features=features,
            )

            if scoring_result is None:
                continue

            candidates.append(
                (features, scoring_result)
            )

        return candidates

    @staticmethod
    def _has_usable_features(features):
        return bool(
            features.tokens
            or features.topics
            or features.keywords
            or features.learning_objectives
        )

    def _filter_candidates(
        self,
        candidates,
        request: ContentRecommendationRequest,
        interactions,
    ):
        from app.services.recommendations.content_based.candidate_models import (
            ContentBasedCandidate,
        )

        from app.services.recommendations.content_based.filtering_service import (
            ContentFilterCriteria,
        )

        wrapped_candidates = []

        for features, scoring_result in candidates:
            wrapped_candidates.append(
                ContentBasedCandidate(
                    features=features,
                    scoring_result=scoring_result,
                    is_active=True,
                    quality_score=None,
                )
            )

        criteria = ContentFilterCriteria(
            resource_types=tuple(request.resource_types),
            subjects=(
                (request.subject,)
                if request.subject
                else ()
            ),
            languages=(
                (request.language,)
                if request.language
                else ()
            ),
            difficulty_levels=(
                (request.difficulty_level,)
                if request.difficulty_level
                else ()
            ),
            minimum_score=0.0,
            excluded_resource_ids=(),
            exclude_inactive=True,
        )

        return self.filtering_service.filter(
            candidates=wrapped_candidates,
            criteria=criteria,
        )

    def _rank_candidates(
        self,
        candidates,
        limit: int,
    ):
        return self.ranking_service.rank(
            candidates=candidates,
            limit=limit,
        )

    @staticmethod
    def _build_recommendation_item(
        ranked,
        similarity_results,
    ):
        candidate = ranked.candidate

        features = candidate.features
        score = candidate.scoring_result

        similarity_by_id = {
            result.resource_id: result
            for result in similarity_results
        }

        similarity_result = similarity_by_id.get(
            features.resource_id
        )

        similarity_score = (
            similarity_result.similarity_score
            if similarity_result is not None
            else 0.0
        )

        resource = (
            similarity_result.resource
            if similarity_result is not None
            else None
        )

        return ContentRecommendationItem(
            resource_id=features.resource_id,
            title=features.title,
            description=getattr(
                resource,
                "description",
                None,
            ),
            resource_type=features.resource_type,
            subject=features.subject,
            language=features.language,
            difficulty_level=features.difficulty,
            url=getattr(
                resource,
                "url",
                None,
            ),
            similarity_score=similarity_score,
            content_score=score.score,
            confidence=score.confidence,
            matched_topics=list(score.matched_topics),
            matched_keywords=list(score.matched_keywords),
            matched_objectives=list(score.matched_objectives),
            explanation=(
                "Recommended based on the relevance of "
                "the resource's content to your learning query."
            ),
        )

    def recommend(
        self,
        db: Session,
        user_id: UUID,
        request: ContentRecommendationRequest,
    ) -> ContentRecommendationResponse:

        if not isinstance(user_id, UUID):
            raise ValueError("Invalid user ID.")

        if not isinstance(
            request,
            ContentRecommendationRequest,
        ):
            raise TypeError(
                "Expected ContentRecommendationRequest."
            )

        try:
            interactions = self._load_interactions(
                db=db,
                user_id=user_id,
            )

            context = self._build_context(
                db=db,
                user_id=user_id,
                request=request,
            )

            similarity_results = self._retrieve_candidates(
                db=db,
                context=context,
                request=request,
            )

            scored_candidates = self._score_candidates(
                similarity_results=similarity_results,
                context=context,
            )

            filtered_candidates = self._filter_candidates(
                candidates=scored_candidates,
                request=request,
                interactions=interactions,
            )

            ranked_candidates = self._rank_candidates(
                candidates=filtered_candidates,
                limit=request.limit,
            )

            recommendations = [
                self._build_recommendation_item(
                    ranked=ranked,
                    similarity_results=similarity_results,
                )
                for ranked in ranked_candidates
            ]

            return ContentRecommendationResponse(
                query=request.query,
                total_results=len(recommendations),
                recommendations=recommendations,
                algorithm_version=self.ALGORITHM_VERSION,
            )

        except Exception as exc:
            raise ContentRecommendationServiceError(
                "Content recommendation generation failed."
            ) from exc

from __future__ import annotations

import inspect

from collections.abc import Iterable
from uuid import UUID

from app.schemas.recommendations import (

)

from app.services.recommendations.context_builder import (
    LearningContextBuilder,
)

class ContentQueryContextError(Exception):
    """Base exception for query-context operations."""


class EmptyLearningIntentError(ContentQueryContextError):
    """Raised when no usable learning intent exists."""


class QueryEmbeddingError(ContentQueryContextError):
    """Raised when query embedding generation fails."""


class ContentQueryContextService:
    """
    Builds a query-aware content recommendation context.

    Responsibilities:
    - Build the user's learning context.
    - Construct a meaningful semantic query.
    - Generate a query embedding.
    - Resolve effective search filters.
    """

    DEFAULT_DIMENSIONS = 1536

    def __init__(
        self,
        embedding_service,
        context_builder=None,
        expected_dimensions=DEFAULT_DIMENSIONS,
    ):

        if embedding_service is None:

            raise TypeError(
                "embedding_service is required."
            )

        if (
            isinstance(expected_dimensions, bool)
            or not isinstance(expected_dimensions, int)
            or expected_dimensions <= 0
        ):

            raise ValueError(
                "expected_dimensions must be a positive integer."
            )

        self.embedding_service = embedding_service

        self.context_builder = (
            context_builder
            or LearningContextBuilder()
        )

        self.expected_dimensions = expected_dimensions

    @staticmethod
    def _normalize_text(value):

        if value is None:
            return ""

        if not isinstance(value, str):
            return ""

        return " ".join(value.split())

    @staticmethod
    def _unique(values: Iterable[str]):

        result = []

        seen = set()

        for value in values:

            normalized = (
                ContentQueryContextService._normalize_text(
                    value
                )
            )

            if not normalized:
                continue

            key = normalized.casefold()

            if key not in seen:

                seen.add(key)

                result.append(normalized)

        return result

    def build_query_text(
        self,
        request: RecommendationRequest,
        learning_context,
    ) -> str:

        if not isinstance(
            request,
            RecommendationRequest,
        ):

            raise TypeError(
                "Expected RecommendationRequest."
            )

        query_parts = []

        explicit_query = self._normalize_text(
            request.query
        )

        if explicit_query:
            query_parts.append(explicit_query)

        subject = self._normalize_text(
            request.subject
        )

        if subject:
            query_parts.append(
                f"Subject: {subject}"
            )

        learning_goals = self._unique(
            learning_context.learning_goals
        )

        if learning_goals:

            query_parts.append(
                "Learning goals: "
                + ", ".join(learning_goals)
            )

        preferred_subjects = self._unique(
            learning_context.preferred_subjects
        )

        if preferred_subjects:

            query_parts.append(
                "Preferred subjects: "
                + ", ".join(preferred_subjects)
            )

        query_text = "\n".join(
            self._unique(query_parts)
        )

        if not query_text:

            raise EmptyLearningIntentError(
                "No query, subject, learning goal or preferred subject "
                "is available to generate a content-based recommendation."
            )

        return query_text

    def build_filters(
        self,
        request: RecommendationRequest,
        learning_context,
    ) -> ContentQueryFilters:

        resource_types = tuple(
            resource_type.value
            if hasattr(resource_type, "value")
            else str(resource_type)
            for resource_type in (
                request.resource_types or []
            )
        )

        language = (
            request.language
            or learning_context.preferred_language
        )

        return ContentQueryFilters(
            subject=request.subject,
            language=language,
            difficulty_level=(
                request.difficulty_level.value
                if request.difficulty_level is not None
                and hasattr(request.difficulty_level, "value")
                else (
                    str(request.difficulty_level)
                    if request.difficulty_level is not None
                    else None
                )
            ),
            resource_types=resource_types,
            preferred_subjects=tuple(
                learning_context.preferred_subjects
            ),
            learning_goals=tuple(
                learning_context.learning_goals
            ),
            exclude_completed=request.exclude_completed,
        )

    async def _embed_query(self, query_text):

        try:

            result = self.embedding_service.embed_query(
                query_text
            )

            if inspect.isawaitable(result):
                result = await result

        except Exception as exc:

            raise QueryEmbeddingError(
                "Unable to generate the query embedding."
            ) from exc

        if result is None:

            raise QueryEmbeddingError(
                "Embedding provider returned no result."
            )

        vector = getattr(
            result,
            "vector",
            None,
        )

        if vector is None:

            raise QueryEmbeddingError(
                "Embedding result does not contain a vector."
            )

        dimensions = getattr(
            result,
            "dimensions",
            None,
        )

        provider = getattr(
            result,
            "provider",
            None,
        )

        model = getattr(
            result,
            "model",
            None,
        )

        if dimensions != self.expected_dimensions:

            raise QueryEmbeddingError(
                "Query embedding dimensions do not match configuration."
            )

        if len(vector) != self.expected_dimensions:

            raise QueryEmbeddingError(
                "Query embedding vector has an invalid length."
            )

        if not provider or not model:

            raise QueryEmbeddingError(
                "Embedding provider and model metadata are required."
            )

        return (
            tuple(float(value) for value in vector),
            provider,
            model,
            dimensions,
        )

    async def build(
        self,
        user,
        request: RecommendationRequest,
        interactions=(),
    ) -> ContentQueryContext:

        if user is None:

            raise TypeError(
                "An authenticated user object is required."
            )

        if not isinstance(
            request,
            RecommendationRequest,
        ):

            raise TypeError(
                "Expected RecommendationRequest."
            )

        user_id = getattr(
            user,
            "id",
            None,
        )

        if not isinstance(user_id, UUID):

            raise TypeError(
                "User must have a valid UUID id."
            )

        try:

            learning_context = self.context_builder.build(
                user=user,
                preferences=getattr(
                    user,
                    "preferences",
                    None,
                ),
                request=request,
                interactions=interactions,
            )

            query_text = self.build_query_text(
                request,
                learning_context,
            )

            filters = self.build_filters(
                request,
                learning_context,
            )

            (
                query_embedding,
                provider,
                model,
                dimensions,
            ) = await self._embed_query(
                query_text
            )

            return ContentQueryContext(
                user_id=user_id,
                query_text=query_text,
                query_embedding=query_embedding,
                learning_context=learning_context,
                filters=filters,
                embedding_provider=provider,
                embedding_model=model,
                embedding_dimensions=dimensions,
            )

        except (
            EmptyLearningIntentError,
            QueryEmbeddingError,
            QueryContextValidationError,
        ):

            raise

        except Exception as exc:

            raise ContentQueryContextError(
                "Failed to construct content recommendation context."
            ) from exc

from __future__ import annotations

import math

from typing import Sequence
from uuid import UUID

from sqlalchemy import select
from sqlalchemy.orm import Session

from app.core.embedding_config import (
    EmbeddingSettings,
    get_embedding_settings,
)

from app.models.resource_embedding import (
    ResourceEmbedding,
    RESOURCE_EMBEDDING_PIPELINE_VERSION,
    VECTOR_DIMENSIONS,
)

from app.models.study_resource import StudyResource

from app.services.recommendations.content_based.similarity_models import (
    ResourceSimilarityResult,
    ResourceComparisonResult,
)


class ResourceSimilarityError(Exception):
    """Base exception for resource similarity operations."""


class ResourceSimilarityValidationError(
    ResourceSimilarityError
):
    """Raised when similarity input is invalid."""


class ResourceEmbeddingNotFoundError(
    ResourceSimilarityError
):
    """Raised when a usable resource embedding is unavailable."""


class ResourceSimilarityService:
    """
    Computes semantic similarity between educational resources.

    Uses PostgreSQL pgvector cosine distance.

    Responsibilities:
    - Validate embedding vectors.
    - Retrieve compatible resource embeddings.
    - Compute cosine similarity.
    - Find nearest educational resources.
    - Compare two individual resources.
    - Apply recommendation-oriented filters.
    """

    def __init__(
        self,
        settings: EmbeddingSettings | None = None,
    ) -> None:

        self.settings = (
            settings or get_embedding_settings()
        )

        if self.settings.dimensions != VECTOR_DIMENSIONS:
            raise ResourceSimilarityValidationError(
                "Configured embedding dimensions must equal "
                f"{VECTOR_DIMENSIONS}."
            )

    # ---------------------------------------------------------
    # Vector validation
    # ---------------------------------------------------------

    @staticmethod
    def validate_vector(
        vector: Sequence[float],
    ) -> list[float]:
        """
        Validate and normalize the input representation
        of a vector for database querying.
        """

        if isinstance(vector, (str, bytes)):
            raise ResourceSimilarityValidationError(
                "Embedding vector must be a numeric sequence."
            )

        if not isinstance(vector, Sequence):
            raise ResourceSimilarityValidationError(
                "Embedding vector must be a sequence."
            )

        if len(vector) != VECTOR_DIMENSIONS:
            raise ResourceSimilarityValidationError(
                f"Embedding vector must contain "
                f"{VECTOR_DIMENSIONS} dimensions."
            )

        validated = []

        for value in vector:

            if isinstance(value, bool):
                raise ResourceSimilarityValidationError(
                    "Boolean values are not valid vector elements."
                )

            try:
                number = float(value)

            except (TypeError, ValueError, OverflowError) as exc:
                raise ResourceSimilarityValidationError(
                    "Embedding vector contains a non-numeric value."
                ) from exc

            if not math.isfinite(number):
                raise ResourceSimilarityValidationError(
                    "Embedding vector contains a non-finite value."
                )

            validated.append(number)

        return validated

    # ---------------------------------------------------------
    # Similarity conversion
    # ---------------------------------------------------------

    @staticmethod
    def distance_to_similarity(
        distance: float,
    ) -> float:
        """
        Convert cosine distance into cosine similarity.

        pgvector cosine distance:
            distance = 1 - cosine_similarity

        Therefore:
            similarity = 1 - distance

        The theoretical cosine similarity range is [-1, 1].
        """

        distance = float(distance)

        if not math.isfinite(distance):
            raise ResourceSimilarityValidationError(
                "Cosine distance must be finite."
            )

        if distance < 0.0 or distance > 2.0:
            raise ResourceSimilarityValidationError(
                "Cosine distance must be between 0 and 2."
            )

        similarity = 1.0 - distance

        return max(-1.0, min(1.0, similarity))

    # ---------------------------------------------------------
    # Embedding identity filters
    # ---------------------------------------------------------

    def _compatible_embedding_filters(self):
        """
        Restrict searches to compatible, current embeddings.
        """

        return (
            ResourceEmbedding.status == "ready",
            ResourceEmbedding.provider == self.settings.provider,
            ResourceEmbedding.model == self.settings.model,
            ResourceEmbedding.dimensions == VECTOR_DIMENSIONS,
            ResourceEmbedding.pipeline_version
            == RESOURCE_EMBEDDING_PIPELINE_VERSION,
        )

    # ---------------------------------------------------------
    # Similar resource search
    # ---------------------------------------------------------

    def find_similar_resources(
        self,
        db: Session,
        query_vector: Sequence[float],
        limit: int = 10,
        min_similarity: float = 0.0,
        exclude_resource_id: UUID | None = None,
        subject: str | None = None,
        resource_type: str | None = None,
        language: str | None = None,
    ) -> list[ResourceSimilarityResult]:
        """
        Find educational resources closest to a query vector.

        Filtering is performed by PostgreSQL before results
        are returned to the application.
        """

        vector = self.validate_vector(query_vector)

        if isinstance(limit, bool) or not isinstance(limit, int):
            raise ResourceSimilarityValidationError(
                "Limit must be an integer."
            )

        if not 1 <= limit <= 100:
            raise ResourceSimilarityValidationError(
                "Limit must be between 1 and 100."
            )

        if isinstance(min_similarity, bool):
            raise ResourceSimilarityValidationError(
                "Minimum similarity must be numeric."
            )

        try:
            min_similarity = float(min_similarity)

        except (TypeError, ValueError, OverflowError) as exc:
            raise ResourceSimilarityValidationError(
                "Minimum similarity must be numeric."
            ) from exc

        if (
            not math.isfinite(min_similarity)
            or not -1.0 <= min_similarity <= 1.0
        ):
            raise ResourceSimilarityValidationError(
                "Minimum similarity must be between -1 and 1."
            )

        distance_expression = (
            ResourceEmbedding.embedding.cosine_distance(
                vector
            )
        )

        statement = (
            select(
                StudyResource,
                ResourceEmbedding.id.label("embedding_id"),
                distance_expression.label("cosine_distance"),
            )
            .join(
                ResourceEmbedding,
                ResourceEmbedding.resource_id
                == StudyResource.id,
            )
            .where(
                StudyResource.is_active.is_(True),
                *self._compatible_embedding_filters(),
            )
        )

        if exclude_resource_id is not None:

            statement = statement.where(
                StudyResource.id != exclude_resource_id
            )

        if subject is not None:

            statement = statement.where(
                StudyResource.subject == subject
            )

        if resource_type is not None:

            statement = statement.where(
                StudyResource.resource_type == resource_type
            )

        if language is not None:

            statement = statement.where(
                StudyResource.language == language
            )

        statement = (
            statement
            .order_by(distance_expression.asc())
            .limit(limit)
        )

        rows = db.execute(statement).all()

        results = []

        for resource, embedding_id, distance in rows:

            distance = float(distance)

            similarity = self.distance_to_similarity(
                distance
            )

            if similarity < min_similarity:
                continue

            results.append(
                ResourceSimilarityResult(
                    resource_id=resource.id,
                    resource=resource,
                    embedding_id=embedding_id,
                    cosine_distance=distance,
                    similarity_score=similarity,
                )
            )

        return results

    # ---------------------------------------------------------
    # Retrieve an individual resource embedding
    # ---------------------------------------------------------

    def get_resource_embedding(
        self,
        db: Session,
        resource_id: UUID,
    ) -> ResourceEmbedding:

        if not isinstance(resource_id, UUID):
            raise ResourceSimilarityValidationError(
                "Resource ID must be a UUID."
            )

        statement = (
            select(ResourceEmbedding)
            .where(
                ResourceEmbedding.resource_id == resource_id,
                *self._compatible_embedding_filters(),
            )
            .order_by(
                ResourceEmbedding.updated_at.desc()
            )
            .limit(1)
        )

        embedding = db.execute(
            statement
        ).scalars().first()

        if embedding is None:
            raise ResourceEmbeddingNotFoundError(
                "No compatible ready embedding exists "
                "for the requested resource."
            )

        if embedding.embedding is None:
            raise ResourceEmbeddingNotFoundError(
                "The resource embedding has no stored vector."
            )

        return embedding

    # ---------------------------------------------------------
    # Resource-to-resource comparison
    # ---------------------------------------------------------

    def compare_resources(
        self,
        db: Session,
        source_resource_id: UUID,
        target_resource_id: UUID,
    ) -> ResourceComparisonResult:
        """
        Compare two resources using their stored embeddings.

        Both resources must have compatible ready embeddings.
        """

        if source_resource_id == target_resource_id:
            raise ResourceSimilarityValidationError(
                "Source and target resources must be different."
            )

        source_embedding = self.get_resource_embedding(
            db,
            source_resource_id,
        )

        target_embedding = self.get_resource_embedding(
            db,
            target_resource_id,
        )

        distance = float(
            db.query(
                ResourceEmbedding.embedding.cosine_distance(
                    list(source_embedding.embedding)
                )
            )
            .filter(
                ResourceEmbedding.id == target_embedding.id
            )
            .scalar()
        )

        similarity = self.distance_to_similarity(
            distance
        )

        return ResourceComparisonResult(
            source_resource_id=source_resource_id,
            target_resource_id=target_resource_id,
            cosine_distance=distance,
            similarity_score=similarity,
        )

    # ---------------------------------------------------------
    # Find resources similar to another resource
    # ---------------------------------------------------------

    def find_similar_to_resource(
        self,
        db: Session,
        resource_id: UUID,
        limit: int = 10,
        min_similarity: float = 0.0,
    ) -> list[ResourceSimilarityResult]:
        """
        Retrieve nearest neighbors of an existing resource.
        """

        source_embedding = self.get_resource_embedding(
            db,
            resource_id,
        )

        return self.find_similar_resources(
            db=db,
            query_vector=list(source_embedding.embedding),
            limit=limit,
            min_similarity=min_similarity,
            exclude_resource_id=resource_id,
        )

import asyncio

from app.core.embedding_config import (

class EmbeddingService:

    def __init__(
        self,
        provider: BaseEmbeddingProvider | None = None,
        settings: EmbeddingSettings | None = None,
    ):
        self.settings = settings or get_embedding_settings()

        if provider is not None:
            self.provider = provider

        elif self.settings.provider == "gemini":
            self.provider = GeminiEmbeddingProvider(
                self.settings
            )

        else:
            raise ValueError(
                "Unsupported embedding provider."
            )

    def _build_request(
        self,
        text: str,
        task: EmbeddingTask,
    ) -> EmbeddingRequest:

        if not isinstance(text, str):
            raise TypeError("Text must be a string.")

        if len(text) > self.settings.max_input_characters:
            raise EmbeddingInputTooLargeError(
                "Input exceeds configured character limit."
            )

        return EmbeddingRequest(
            text=text,
            task=task,
        )

    async def embed_document(
        self,
        text: str
    ) -> EmbeddingResult:

        request = self._build_request(
            text,
            EmbeddingTask.DOCUMENT,
        )

        return await self.provider.embed(request)

    async def embed_query(
        self,
        text: str
    ) -> EmbeddingResult:

        request = self._build_request(
            text,
            EmbeddingTask.QUERY,
        )

        return await self.provider.embed(request)

    async def embed_documents(
        self,
        texts: list[str]
    ) -> list[EmbeddingResult]:

        if not isinstance(texts, list):
            raise TypeError("Texts must be provided as a list.")

        if not texts:
            raise ValueError("Document batch cannot be empty.")

        requests = [
            self._build_request(
                text,
                EmbeddingTask.DOCUMENT,
            )
            for text in texts
        ]

        semaphore = asyncio.Semaphore(
            self.settings.max_concurrent_requests
        )

        async def process(request):
            async with semaphore:
                return await self.provider.embed(request)

        return await asyncio.gather(
            *(process(request) for request in requests)
        )

    async def close(self) -> None:
        await self.provider.close()
