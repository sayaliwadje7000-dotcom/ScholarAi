class EmbeddingPersistenceService:

    def __init__(
        self,
        settings: EmbeddingSettings | None = None,
    ) -> None:
        self.settings = settings or get_embedding_settings()

    def _validate_outcome(
        self,
        outcome: ChunkEmbeddingOutcome,
    ) -> None:

        if not outcome.chunk_id:
            raise EmbeddingPersistenceValidationError(
                "Chunk ID cannot be empty."
            )

        if not outcome.content_hash:
            raise EmbeddingPersistenceValidationError(
                "Content hash cannot be empty."
            )

        if outcome.status not in {"ready", "failed"}:
            raise EmbeddingPersistenceValidationError(
                "Unsupported embedding status."
            )

        if outcome.status == "failed":
            return

        embedding = outcome.embedding

        if embedding is None:
            raise EmbeddingPersistenceValidationError(
                "Ready embedding must contain a vector."
            )

        if embedding.dimensions != DATABASE_VECTOR_DIMENSIONS:
            raise EmbeddingPersistenceValidationError(
                "Embedding dimensions do not match database storage."
            )

        if len(embedding.vector) != DATABASE_VECTOR_DIMENSIONS:
            raise EmbeddingPersistenceValidationError(
                "Invalid embedding vector length."
            )

        if embedding.provider != self.settings.provider:
            raise EmbeddingPersistenceValidationError(
                "Embedding provider mismatch."
            )

        if embedding.model != self.settings.model:
            raise EmbeddingPersistenceValidationError(
                "Embedding model mismatch."
            )

        if not all(
            math.isfinite(float(value))
            for value in embedding.vector
        ):
            raise EmbeddingPersistenceValidationError(
                "Embedding contains non-finite values."
            )

    def _identity(
        self,
        outcome: ChunkEmbeddingOutcome,
    ) -> dict:

        embedding = outcome.embedding

        return {
            "chunk_id": outcome.chunk_id,
            "content_hash": outcome.content_hash,
            "provider": (
                embedding.provider
                if embedding is not None
                else self.settings.provider
            ),
            "model": (
                embedding.model
                if embedding is not None
                else self.settings.model
            ),
            "model_version": (
                embedding.model
                if embedding is not None
                else self.settings.model
            ),
            "dimensions": (
                embedding.dimensions
                if embedding is not None
                else DATABASE_VECTOR_DIMENSIONS
            ),
            "pipeline_version": PIPELINE_VERSION,
        }

    @staticmethod
    def _get_first(
        db: Session,
        statement,
    ):
        """Execute a SELECT and return its first scalar result."""
        return db.execute(statement).scalars().first()

    def _get_existing_embedding(
        self,
        db: Session,
        identity: dict,
    ) -> ChunkEmbedding | None:

        statement = select(ChunkEmbedding).where(
            ChunkEmbedding.chunk_id == identity["chunk_id"],
            ChunkEmbedding.content_hash == identity["content_hash"],
            ChunkEmbedding.provider == identity["provider"],
            ChunkEmbedding.model == identity["model"],
            ChunkEmbedding.model_version == identity["model_version"],
            ChunkEmbedding.dimensions == identity["dimensions"],
            ChunkEmbedding.pipeline_version == identity["pipeline_version"],
        )

        return self._get_first(db, statement)

    @staticmethod
    def _vector_values(
        embedding,
    ) -> list[float] | None:

        if embedding is None:
            return None

        return [
            float(value)
            for value in embedding
        ]

    def persist(
        self,
        db: Session,
        outcomes: Sequence[ChunkEmbeddingOutcome],
    ) -> EmbeddingPersistenceResult:

        if isinstance(outcomes, (str, bytes)):
            raise EmbeddingPersistenceValidationError(
                "Outcomes must be a sequence."
            )

        counters = {
            "inserted": 0,
            "updated": 0,
            "unchanged": 0,
            "failed": 0,
        }

        try:
            for outcome in outcomes:

                self._validate_outcome(outcome)

                chunk_statement = select(DocumentChunk).where(
                    DocumentChunk.chunk_id == outcome.chunk_id
                )

                chunk = self._get_first(
                    db,
                    chunk_statement,
                )

                if chunk is None:
                    raise EmbeddingPersistenceValidationError(
                        f"Chunk not found: {outcome.chunk_id}"
                    )

                if chunk.content_hash != outcome.content_hash:
                    raise EmbeddingPersistenceValidationError(
                        "Chunk content hash mismatch. "
                        "The chunk may have changed since embedding."
                    )

                identity = self._identity(outcome)

                existing = self._get_existing_embedding(
                    db,
                    identity,
                )

                if outcome.status == "failed":
                    counters["failed"] += 1

                # Do not replace a valid vector with a transient failure.
                if (
                    existing is not None
                    and existing.status == "ready"
                    and outcome.status == "failed"
                ):
                    counters["unchanged"] += 1
                    continue

                vector = (
                    list(outcome.embedding.vector)
                    if outcome.embedding is not None
                    else None
                )

                values = {
                    **identity,
                    "status": outcome.status,
                    "embedding": vector,
                    "error_code": (
                        outcome.error_code
                        if outcome.status == "failed"
                        else None
                    ),
                }

                if existing is not None:

                    current_vector = self._vector_values(
                        existing.embedding
                    )

                    unchanged = (
                        existing.status == values["status"]
                        and current_vector == values["embedding"]
                        and existing.error_code == values["error_code"]
                    )

                    if unchanged:
                        counters["unchanged"] += 1
                        continue

                    for field, value in values.items():
                        setattr(existing, field, value)

                    counters["updated"] += 1

                else:
                    db.add(
                        ChunkEmbedding(**values)
                    )
                    counters["inserted"] += 1

            db.flush()
            db.commit()

        except Exception:
            db.rollback()
            raise

        return EmbeddingPersistenceResult(
            total=len(outcomes),
            inserted=counters["inserted"],
            updated=counters["updated"],
            unchanged=counters["unchanged"],
            failed=counters["failed"],
        )

class EmbeddingPipelineError(Exception):
    """Base exception for embedding pipeline failures."""


class EmbeddingPipelineInputError(EmbeddingPipelineError):
    """Raised when chunk input is invalid."""


class EmbeddingPipeline:
    """
    Generates embeddings for persistent document chunks.

    Responsibilities:
    - Validate chunk identity and content.
    - Verify deterministic content hashes.
    - Process chunks in bounded batches.
    - Reuse the existing EmbeddingService.
    - Track individual successes and failures.
    - Validate embedding metadata and dimensions.

    Database persistence is intentionally excluded.
    """

    def __init__(
        self,
        embedding_service: EmbeddingService,
        batch_size: int = 32,
    ):
        if batch_size <= 0:
            raise ValueError("Batch size must be positive.")

        self.embedding_service = embedding_service
        self.batch_size = batch_size

        self.settings = embedding_service.settings

    @staticmethod
    def _validate_chunks(chunks: Sequence) -> list:
        if isinstance(chunks, (str, bytes)):
            raise EmbeddingPipelineInputError(
                "Chunks must be a sequence of chunk objects."
            )

        if not isinstance(chunks, Sequence):
            raise EmbeddingPipelineInputError(
                "Chunks must be a sequence."
            )

        validated = list(chunks)

        seen_ids = set()

        for chunk in validated:
            chunk_id = getattr(chunk, "chunk_id", None)
            text = getattr(chunk, "text", None)
            content_hash = getattr(chunk, "content_hash", None)

            if not isinstance(chunk_id, str) or not chunk_id.strip():
                raise EmbeddingPipelineInputError(
                    "Every chunk must have a valid chunk_id."
                )

            if chunk_id in seen_ids:
                raise EmbeddingPipelineInputError(
                    f"Duplicate chunk identifier: {chunk_id}"
                )

            seen_ids.add(chunk_id)

            if not isinstance(text, str) or not text.strip():
                raise EmbeddingPipelineInputError(
                    f"Chunk {chunk_id} has empty or invalid text."
                )

            if not isinstance(content_hash, str):
                raise EmbeddingPipelineInputError(
                    f"Chunk {chunk_id} has an invalid content hash."
                )

            expected_hash = compute_content_hash(text)

            if content_hash != expected_hash:
                raise EmbeddingPipelineInputError(
                    f"Content hash mismatch for chunk {chunk_id}."
                )

        return validated

    def _validate_embedding(
        self,
        result: EmbeddingResult,
    ) -> None:

        if not isinstance(result, EmbeddingResult):
            raise TypeError(
                "Embedding service returned an invalid result."
            )

        if result.dimensions != self.settings.dimensions:
            raise ValueError(
                "Embedding dimensions do not match configured dimensions."
            )

        if result.model != self.settings.model:
            raise ValueError(
                "Embedding model does not match configured model."
            )

        if result.provider != self.settings.provider:
            raise ValueError(
                "Embedding provider does not match configured provider."
            )

        if len(result.vector) != self.settings.dimensions:
            raise ValueError(
                "Embedding vector length is invalid."
            )

    async def _process_chunk(
        self,
        chunk,
    ) -> ChunkEmbeddingOutcome:

        chunk_id = chunk.chunk_id
        content_hash = chunk.content_hash

        try:
            embedding = await self.embedding_service.embed_document(
                chunk.text
            )

            self._validate_embedding(embedding)

            return ChunkEmbeddingOutcome(
                chunk_id=chunk_id,
                content_hash=content_hash,
                status="ready",
                embedding=embedding,
            )

        except asyncio.CancelledError:
            raise

        except Exception as exc:
            return ChunkEmbeddingOutcome(
                chunk_id=chunk_id,
                content_hash=content_hash,
                status="failed",
                error_code=type(exc).__name__,
            )

    async def generate(
        self,
        chunks: Sequence,
    ) -> EmbeddingPipelineResult:
        """
        Generate embeddings for a collection of document chunks.

        Results preserve the original input order.
        Individual chunk failures do not terminate the pipeline.
        """

        started_at = time.perf_counter()

        validated_chunks = self._validate_chunks(chunks)

        if not validated_chunks:
            return EmbeddingPipelineResult(
                total_chunks=0,
                successful_chunks=0,
                failed_chunks=0,
                outcomes=(),
                provider=self.settings.provider,
                model=self.settings.model,
                dimensions=self.settings.dimensions,
                duration_seconds=0.0,
            )

        outcomes = []

        for start in range(
            0,
            len(validated_chunks),
            self.batch_size,
        ):
            batch = validated_chunks[
                start:start + self.batch_size
            ]

            batch_results = await asyncio.gather(
                *(
                    self._process_chunk(chunk)
                    for chunk in batch
                )
            )

            outcomes.extend(batch_results)

        successful = sum(
            outcome.status == "ready"
            for outcome in outcomes
        )

        failed = len(outcomes) - successful

        duration = time.perf_counter() - started_at

        return EmbeddingPipelineResult(
            total_chunks=len(validated_chunks),
            successful_chunks=successful,
            failed_chunks=failed,
            outcomes=tuple(outcomes),
            provider=self.settings.provider,
            model=self.settings.model,
            dimensions=self.settings.dimensions,
            duration_seconds=round(duration, 6),
        )

        return await self.provider.embed(request)

        self,

        if not texts:
            raise ValueError("Document batch cannot be empty.")

            self._build_request(
                text,

class GeminiEmbeddingProvider(BaseEmbeddingProvider):
    """Gemini Embedding 2 provider."""

    def __init__(
        self,
        settings: EmbeddingSettings,
        client=None,
    ):
        self.settings = settings

        if client is None:
            if not settings.api_key:
                raise EmbeddingAuthenticationError(
                    "GEMINI_API_KEY is not configured."
                )

            self.client = genai.Client(
                api_key=settings.api_key,
                http_options=types.HttpOptions(
                    timeout=int(
                        settings.request_timeout_seconds * 1000
                    ),
                    retry_options=types.HttpRetryOptions(
                        attempts=1,
                    ),
                ),
            )
        else:
            self.client = client

        self._semaphore = asyncio.Semaphore(
            settings.max_concurrent_requests
        )

    @staticmethod
    def _prepare_input(request: EmbeddingRequest) -> str:
        """Prepare model-specific retrieval instructions."""

        if request.task == EmbeddingTask.DOCUMENT:
            return f"title: none | text: {request.text}"

        if request.task == EmbeddingTask.QUERY:
            return (
                "task: search result | query: "
                f"{request.text}"
            )

        raise EmbeddingValidationError(
            "Unsupported embedding task."
        )

    @staticmethod
    def _classify_exception(exc: Exception):
        """Map SDK failures to application exceptions."""

        if isinstance(exc, (TimeoutError, asyncio.TimeoutError)):
            return EmbeddingTimeoutError

        if isinstance(exc, errors.APIError):
            status_code = exc.code
            message = (exc.message or "").lower()
            status = (exc.status or "").upper()

            if status_code in (401, 403):
                return EmbeddingAuthenticationError

            if status_code == 429:
                return EmbeddingRateLimitError

            if status_code in (408, 504):
                return EmbeddingTimeoutError

            if (
                status_code in (413,)
                or "token limit" in message
                or "input too long" in message
                or "too many tokens" in message
            ):
                return EmbeddingInputTooLargeError

            if status_code is not None and 500 <= status_code <= 599:
                return EmbeddingProviderError

            if status_code is not None and 400 <= status_code <= 499:
                return EmbeddingProviderError

            if status in ("RESOURCE_EXHAUSTED",):
                return EmbeddingRateLimitError

            if status in ("UNAUTHENTICATED", "PERMISSION_DENIED"):
                return EmbeddingAuthenticationError

        return EmbeddingProviderError

    @staticmethod
    def _is_retryable(
        error: Exception,
        original_exception: Exception,
    ) -> bool:
        """Only retry transient failures."""

        if isinstance(
            error,
            (
                EmbeddingRateLimitError,
                EmbeddingTimeoutError,
            ),
        ):
            return True

        if isinstance(original_exception, errors.ServerError):
            return True

        return False

    def _validate_vector(self, values) -> tuple[float, ...]:
        """Validate embedding dimensions and numerical integrity."""

        if values is None:
            raise EmbeddingValidationError(
                "Provider returned an empty vector."
            )

        try:
            vector = tuple(
                float(value) for value in values
            )
        except (TypeError, ValueError, OverflowError) as exc:
            raise EmbeddingValidationError(
                "Provider returned invalid vector values."
            ) from exc

        if len(vector) != self.settings.dimensions:
            raise EmbeddingValidationError(
                f"Expected {self.settings.dimensions} dimensions, "
                f"received {len(vector)}."
            )

        if not all(math.isfinite(value) for value in vector):
            raise EmbeddingValidationError(
                "Vector contains non-finite values."
            )

        return vector

    async def embed(
        self,
        request: EmbeddingRequest,
    ) -> EmbeddingResult:
        """Generate one embedding with bounded retries."""

        if len(request.text) > self.settings.max_input_characters:
            raise EmbeddingInputTooLargeError(
                "Input exceeds configured character limit."
            )

        prepared_text = self._prepare_input(request)

        async with self._semaphore:

            for attempt in range(
                self.settings.max_retries + 1
            ):
                try:
                    response = (
                        await self.client.aio.models.embed_content(
                            model=self.settings.model,
                            contents=prepared_text,
                            config=types.EmbedContentConfig(
                                output_dimensionality=(
                                    self.settings.dimensions
                                )
                            ),
                        )
                    )

                    embeddings = getattr(
                        response,
                        "embeddings",
                        None,
                    )

                    if (
                        not embeddings
                        or len(embeddings) != 1
                    ):
                        raise EmbeddingValidationError(
                            "Unexpected embedding response count."
                        )

                    vector = self._validate_vector(
                        embeddings[0].values
                    )

                    return EmbeddingResult(
                        vector=vector,
                        model=self.settings.model,
                        provider="gemini",
                        dimensions=len(vector),
                    )

                except asyncio.CancelledError:
                    raise

                except EmbeddingValidationError:
                    raise

                except Exception as exc:
                    error_type = self._classify_exception(exc)

                    error = error_type(
                        "Gemini embedding request failed."
                    )

                    if (
                        not self._is_retryable(error, exc)
                        or attempt >= self.settings.max_retries
                    ):
                        raise error from exc

                    delay = min(
                        self.settings.retry_max_delay_seconds,
                        self.settings.retry_base_delay_seconds
                        * (2 ** attempt),
                    )

                    delay *= random.uniform(0.75, 1.25)

                    await asyncio.sleep(delay)

        raise EmbeddingProviderError(
            "Embedding generation failed."
        )

        """Close the asynchronous SDK client."""

        aio_client = getattr(
            self.client,
            "aio",
            None,
        )

        if aio_client is None:

            "aclose",
            None,
        )

        if close is not None:
            result = close()

            if hasattr(result, "__await__"):
                await result

def _percentile(values: list[float], percentile: float) -> float:
    """Calculate percentile using linear interpolation."""

    ordered = sorted(values)

    if len(ordered) == 1:
        return ordered[0]

    position = (len(ordered) - 1) * percentile
    lower_index = int(position)
    upper_index = min(lower_index + 1, len(ordered) - 1)

    fraction = position - lower_index

    return (
        ordered[lower_index] * (1 - fraction)
        + ordered[upper_index] * fraction
    )

    warmup_iterations: int,
    total_elapsed_seconds: float,
) -> BenchmarkResult:
    total_latency = sum(latencies)

    return BenchmarkResult(
        iterations=len(latencies),
        warmup_iterations=warmup_iterations,
        mean_latency_ms=statistics.mean(latencies),
        median_latency_ms=statistics.median(latencies),
        p95_latency_ms=_percentile(latencies, 0.95),
        minimum_latency_ms=min(latencies),
        maximum_latency_ms=max(latencies),
        throughput_per_second=(
            len(latencies) / total_elapsed_seconds
            if total_elapsed_seconds > 0
            else 0.0
        ),
    )


def benchmark_callable(
    function: Callable[[], Any],
    iterations: int = 100,
    warmup_iterations: int = 5,
) -> BenchmarkResult:
    """
    Benchmark a synchronous callable.

    Awaitable results are rejected. Use benchmark_async_callable
    for asynchronous functions.
    """

    _validate_iterations(iterations, warmup_iterations)

    if not callable(function):
        raise TypeError("function must be callable.")

    for _ in range(warmup_iterations):
        result = function()

        if inspect.isawaitable(result):
            if inspect.iscoroutine(result):
                result.close()

            raise TypeError(
                "An asynchronous callable requires benchmark_async_callable."
            )

    latencies: list[float] = []

    total_start = time.perf_counter()

    for _ in range(iterations):
        start = time.perf_counter()

        result = function()

        if inspect.isawaitable(result):
            if inspect.iscoroutine(result):
                result.close()

            raise TypeError(
                "An asynchronous callable requires benchmark_async_callable."
            )

        elapsed = time.perf_counter() - start
        latencies.append(elapsed * 1000)

    total_elapsed = time.perf_counter() - total_start

    return _build_result(
        latencies,
        warmup_iterations,
        total_elapsed,
    )


async def benchmark_async_callable(
    function: Callable[[], Any],
    iterations: int = 100,
    warmup_iterations: int = 5,
) -> BenchmarkResult:
    """
    Benchmark an asynchronous callable sequentially.

    Supports both async functions and synchronous functions
    returning awaitable results.
    """

    _validate_iterations(iterations, warmup_iterations)

    if not callable(function):
        raise TypeError("function must be callable.")

    async def execute() -> None:
        result = function()

        if inspect.isawaitable(result):
            await result

    for _ in range(warmup_iterations):
        await execute()

    latencies: list[float] = []

    total_start = time.perf_counter()

    for _ in range(iterations):
        start = time.perf_counter()

        await execute()

        elapsed = time.perf_counter() - start
        latencies.append(elapsed * 1000)

    total_elapsed = time.perf_counter() - total_start

    return _build_result(
        latencies,
        warmup_iterations,
        total_elapsed,
    )

from __future__ import annotations

from dataclasses import dataclass
from typing import Hashable, Mapping, Sequence


@dataclass(frozen=True, slots=True)
class RetrievalMetrics:
    precision_at_k: float
    recall_at_k: float
    mean_reciprocal_rank: float
    relevant_retrieved: int
    retrieved_count: int
    relevant_count: int


@dataclass(frozen=True, slots=True)
class AggregateRetrievalMetrics:
    precision_at_k: float
    recall_at_k: float
    mean_reciprocal_rank: float
    query_count: int


def evaluate_retrieval(
    retrieved_ids: Sequence[Hashable],
    relevant_ids: Sequence[Hashable],
    k: int = 5,
) -> RetrievalMetrics:
    """
    Evaluate a single ranked retrieval result.

    Precision@K = relevant retrieved in top K / K
    Recall@K = relevant retrieved in top K / total relevant
    MRR = reciprocal rank of the first relevant result.
    """

    if isinstance(k, bool) or not isinstance(k, int) or k <= 0:
        raise ValueError("k must be a positive integer.")

    if len(set(retrieved_ids)) != len(retrieved_ids):
        raise ValueError("retrieved_ids must not contain duplicates.")

    relevant_set = set(relevant_ids)

    if not relevant_set:
        raise ValueError("relevant_ids must contain at least one relevant item.")

    ranked_results = list(retrieved_ids[:k])

    relevant_retrieved = sum(
        1 for item in ranked_results if item in relevant_set
    )

    precision = relevant_retrieved / k
    recall = relevant_retrieved / len(relevant_set)

    reciprocal_rank = 0.0

    for rank, item in enumerate(ranked_results, start=1):
        if item in relevant_set:
            reciprocal_rank = 1.0 / rank
            break

    return RetrievalMetrics(
        precision_at_k=precision,
        recall_at_k=recall,
        mean_reciprocal_rank=reciprocal_rank,
        relevant_retrieved=relevant_retrieved,
        retrieved_count=len(ranked_results),
        relevant_count=len(relevant_set),
    )


def evaluate_retrieval_batch(
    retrieved_by_query: Mapping[Hashable, Sequence[Hashable]],
    relevant_by_query: Mapping[Hashable, Sequence[Hashable]],
    k: int = 5,
) -> AggregateRetrievalMetrics:
    """
    Evaluate multiple queries using macro-averaged metrics.

    Every evaluated query must have at least one relevant item.
    """

    if not retrieved_by_query:
        raise ValueError("At least one retrieval query is required.")

    if set(retrieved_by_query) != set(relevant_by_query):
        raise ValueError(
            "Retrieved and relevant query IDs must match exactly."
        )

    results = [
        evaluate_retrieval(
            retrieved_ids=retrieved_by_query[query_id],
            relevant_ids=relevant_by_query[query_id],
            k=k,
        )
        for query_id in retrieved_by_query
    ]

    count = len(results)

    return AggregateRetrievalMetrics(
        precision_at_k=sum(
            result.precision_at_k for result in results
        ) / count,
        recall_at_k=sum(
            result.recall_at_k for result in results
        ) / count,
        mean_reciprocal_rank=sum(
            result.mean_reciprocal_rank for result in results
        ) / count,
        query_count=count,
    )

from __future__ import annotations

import math
import uuid

from collections.abc import Sequence
from dataclasses import dataclass

from sqlalchemy import and_, select

from app.models.chunk_embedding import ChunkEmbedding

DATABASE_VECTOR_DIMENSIONS = 1536

DEFAULT_SEARCH_LIMIT = 10
MAX_SEARCH_LIMIT = 100


class SemanticSearchError(Exception):
    """Base exception for semantic search operations."""


class SemanticSearchValidationError(SemanticSearchError):
    """Raised when search input or filters are invalid."""


@dataclass(frozen=True)
class SemanticSearchResult:
    chunk_id: str
    document_id: uuid.UUID
    chunk_index: int
    text: str

    page_numbers: list[int]
    start_page: int | None
    end_page: int | None

    similarity: float
    distance: float

    provider: str
    model: str
    dimensions: int
    pipeline_version: str
    chunking_version: str


class SemanticSearchService:
    """
    PostgreSQL pgvector semantic retrieval service.

    Uses cosine distance to identify the most relevant
    document chunks.

    Supports metadata filtering by:
    - Document ID
    - Multiple document IDs
    - Chunking version
    - Embedding provider
    - Embedding model
    - Pipeline version
    """

    def __init__(
        self,
        vector_dimensions: int = DATABASE_VECTOR_DIMENSIONS,
        default_limit: int = DEFAULT_SEARCH_LIMIT,
        max_limit: int = MAX_SEARCH_LIMIT,
    ) -> None:

        if vector_dimensions != DATABASE_VECTOR_DIMENSIONS:
            raise SemanticSearchValidationError(
                "Configured dimensions must match the database "
                f"VECTOR({DATABASE_VECTOR_DIMENSIONS}) column."
            )

        if default_limit < 1:
            raise SemanticSearchValidationError(
                "Default search limit must be positive."
            )

        if max_limit < default_limit:
            raise SemanticSearchValidationError(
                "Maximum limit cannot be smaller than default limit."
            )

        self.vector_dimensions = vector_dimensions
        self.default_limit = default_limit
        self.max_limit = max_limit

    def _validate_vector(
        self,
        query_vector: Sequence[float],
    ) -> list[float]:

        if isinstance(query_vector, (str, bytes)):
            raise SemanticSearchValidationError(
                "Query embedding must be a numeric sequence."
            )

        if not isinstance(query_vector, Sequence):
            raise SemanticSearchValidationError(
                "Query embedding must be a sequence."
            )

        if len(query_vector) != self.vector_dimensions:
            raise SemanticSearchValidationError(
                f"Expected {self.vector_dimensions} dimensions, "
                f"received {len(query_vector)}."
            )

        vector: list[float] = []

        for value in query_vector:

            if isinstance(value, bool) or not isinstance(
                value, (int, float)
            ):
                raise SemanticSearchValidationError(
                    "Embedding contains a non-numeric value."
                )

            if not math.isfinite(value):
                raise SemanticSearchValidationError(
                    "Embedding contains a non-finite value."
                )

            vector.append(float(value))

        return vector

    def _validate_filters(
        self,
        document_id: uuid.UUID | None,
        document_ids: Sequence[uuid.UUID] | None,
        min_similarity: float | None,
    ) -> None:

        if document_id is not None and not isinstance(
            document_id, uuid.UUID
        ):
            raise SemanticSearchValidationError(
                "Document ID must be a UUID."
            )

        if document_ids is not None:

            if isinstance(document_ids, (str, bytes)):
                raise SemanticSearchValidationError(
                    "Document IDs must be a sequence."
                )

            if not isinstance(document_ids, Sequence):
                raise SemanticSearchValidationError(
                    "Document IDs must be a sequence."
                )

            if not document_ids:
                raise SemanticSearchValidationError(
                    "Document IDs cannot be empty."
                )

            if any(
                not isinstance(item, uuid.UUID)
                for item in document_ids
            ):
                raise SemanticSearchValidationError(
                    "Every document ID must be a UUID."
                )

            if len(set(document_ids)) != len(document_ids):
                raise SemanticSearchValidationError(
                    "Duplicate document IDs are not allowed."
                )

        if document_id is not None and document_ids is not None:
            raise SemanticSearchValidationError(
                "Use document_id or document_ids, not both."
            )

        if min_similarity is not None:

            if isinstance(min_similarity, bool) or not isinstance(
                min_similarity, (int, float)
            ):
                raise SemanticSearchValidationError(
                    "Minimum similarity must be numeric."
                )

            if not math.isfinite(min_similarity):
                raise SemanticSearchValidationError(
                    "Minimum similarity must be finite."
                )

            if not -1.0 <= min_similarity <= 1.0:
                raise SemanticSearchValidationError(
                    "Minimum similarity must be between -1 and 1."
                )

    def search(
        self,
        db: Session,
        query_vector: Sequence[float],
        *,
        limit: int | None = None,
        document_id: uuid.UUID | None = None,
        document_ids: Sequence[uuid.UUID] | None = None,
        chunking_version: str | None = None,
        provider: str | None = None,
        model: str | None = None,
        pipeline_version: str | None = None,
        min_similarity: float | None = None,
    ) -> list[SemanticSearchResult]:

        # Validate query embedding.
        vector = self._validate_vector(query_vector)

        # Validate document filters.
        self._validate_filters(
            document_id,
            document_ids,
            min_similarity,
        )

        # Validate result limit.
        if limit is None:
            limit = self.default_limit

        if isinstance(limit, bool) or not isinstance(limit, int):
            raise SemanticSearchValidationError(
                "Search limit must be an integer."
            )

        if not 1 <= limit <= self.max_limit:
            raise SemanticSearchValidationError(
                f"Search limit must be between 1 and {self.max_limit}."
            )

        # Validate metadata filters.
        if chunking_version is not None and (
            not isinstance(chunking_version, str)
            or not chunking_version.strip()
        ):
            raise SemanticSearchValidationError(
                "Chunking version cannot be empty."
            )

        if provider is not None and (
            not isinstance(provider, str) or not provider.strip()
        ):
            raise SemanticSearchValidationError(
                "Provider cannot be empty."
            )

        if model is not None and (
            not isinstance(model, str) or not model.strip()
        ):
            raise SemanticSearchValidationError(
                "Model cannot be empty."
            )

        if pipeline_version is not None and (
            not isinstance(pipeline_version, str)
            or not pipeline_version.strip()
        ):
            raise SemanticSearchValidationError(
                "Pipeline version cannot be empty."
            )

        # Calculate cosine distance using pgvector.
        distance_expression = (
            ChunkEmbedding.embedding
            .cosine_distance(vector)
            .label("distance")
        )

        # Base semantic search query.
        statement = (
            select(
                DocumentChunk,
                ChunkEmbedding,
                distance_expression,
            )
            .join(
                ChunkEmbedding,
                and_(
                    ChunkEmbedding.chunk_id == DocumentChunk.chunk_id,
                    ChunkEmbedding.content_hash
                    == DocumentChunk.content_hash,
                ),
            )
            .where(
                ChunkEmbedding.status == "ready",
                ChunkEmbedding.embedding.is_not(None),
                ChunkEmbedding.dimensions == self.vector_dimensions,
            )
        )

        # Filter by one document.
        if document_id is not None:
            statement = statement.where(
                DocumentChunk.document_id == document_id
            )

        # Filter by multiple documents.
        if document_ids is not None:
            statement = statement.where(
                DocumentChunk.document_id.in_(document_ids)
            )

        # Filter by chunking version.
        if chunking_version is not None:
            statement = statement.where(
                DocumentChunk.chunking_version == chunking_version
            )

        # Filter by embedding provider.
        if provider is not None:
            statement = statement.where(
                ChunkEmbedding.provider == provider
            )

        # Filter by embedding model.
        if model is not None:
            statement = statement.where(
                ChunkEmbedding.model == model
            )

        # Filter by embedding pipeline version.
        if pipeline_version is not None:
            statement = statement.where(
                ChunkEmbedding.pipeline_version == pipeline_version
            )

        # Cosine similarity = 1 - cosine distance.
        if min_similarity is not None:
            statement = statement.where(
                distance_expression <= 1.0 - min_similarity
            )

        # Return the closest matching chunks first.
        statement = (
            statement
            .order_by(distance_expression.asc())
            .limit(limit)
        )

        rows = db.execute(statement).all()

        results: list[SemanticSearchResult] = []

        for chunk, embedding, distance in rows:

            distance = float(distance)
            similarity = 1.0 - distance

            results.append(
                SemanticSearchResult(
                    chunk_id=chunk.chunk_id,
                    document_id=chunk.document_id,
                    chunk_index=chunk.chunk_index,
                    text=chunk.text,
                    page_numbers=chunk.page_numbers or [],
                    start_page=chunk.start_page,
                    end_page=chunk.end_page,
                    similarity=similarity,
                    distance=distance,
                    provider=embedding.provider,
                    model=embedding.model,
                    dimensions=embedding.dimensions,
                    pipeline_version=embedding.pipeline_version,
                    chunking_version=chunk.chunking_version,
                )
            )

        return results
