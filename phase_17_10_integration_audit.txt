@asynccontextmanager
async def lifespan(app: FastAPI):
    """
    Manages application startup and shutdown resources.
    """

    opportunity_registry = OpportunityProviderRegistry()

    app.state.opportunity_provider_registry = (
        opportunity_registry
    )

    try:
        yield

    finally:
        await opportunity_registry.close()

from fastapi import (
    APIRouter,
    Depends,
    HTTPException,
    status,
)

from sqlalchemy.orm import Session

from app.api.dependencies import get_current_user
from app.api.rag_dependencies import get_rag_service

from app.database.session import get_db

from app.models.user import User

from app.schemas.retrieval import RetrievalRequest
from app.schemas.rag import RAGAskResponse

from app.services.rag.rag_service import RAGService


logger = logging.getLogger(__name__)

router = APIRouter()


@router.post(
    "/ask",
    response_model=RAGAskResponse,
    status_code=status.HTTP_200_OK,
    summary="Ask a question using uploaded documents",
    description=(
        "Retrieves relevant document chunks, generates an "
        "evidence-grounded answer, extracts citations, and "
        "returns grounding verification metadata."
    ),
)
async def ask_rag(
    request: RetrievalRequest,
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user),
    rag_service: RAGService = Depends(get_rag_service),
) -> RAGAskResponse:
    """
    Execute the complete RAG pipeline for an authenticated user.
    """

    try:

        result = await rag_service.ask(
            db=db,
            user_id=current_user.id,
            request=request,
        )

        return RAGAskResponse.model_validate(result)

    except HTTPException:
        raise

    except RuntimeError as exc:

        logger.exception(
            "RAG generation failed for user %s.",
            current_user.id,
        )

        raise HTTPException(
            status_code=status.HTTP_503_SERVICE_UNAVAILABLE,
            detail="The RAG service is temporarily unavailable.",
        ) from exc

@lru_cache(maxsize=1)
def get_rag_service() -> RAGService:
    """
    Construct and reuse the RAG service and its dependencies.

    The embedding provider is initialized lazily when the
    first RAG request requires the service.
    """

    embedding_service = EmbeddingService()

    retrieval_service = RetrievalService(
        embedding_service=embedding_service,
    )

    return RAGService(
        retrieval_service=retrieval_service,
    )

from collections.abc import Generator

from sqlalchemy import create_engine
from sqlalchemy.orm import Session, sessionmaker

from app.core.config import settings


if not settings.DATABASE_URL:
    raise RuntimeError(
        "DATABASE_URL is missing. Check your .env configuration."
    )


engine = create_engine(
    settings.DATABASE_URL,
    pool_pre_ping=True,
    pool_recycle=1800,
)


SessionLocal = sessionmaker(
    bind=engine,
    class_=Session,
    autoflush=False,
    autocommit=False,
    expire_on_commit=False,
)


def get_db() -> Generator[Session, None, None]:
    """
    Provides a database session to API endpoints.

    Automatically rolls back unfinished transactions on errors
    and closes the session after use.
    """
    db = SessionLocal()

    try:
        yield db

    except Exception:
        db.rollback()
        raise

    finally:
        db.close()

import os

from dataclasses import dataclass


def _read_provider_name(
    name: str,
    default: str,
) -> str:

    value = os.getenv(name, default).strip().lower()

    if not value:
        raise ValueError(
            f"{name} cannot be empty."
        )

    return value


def _read_fallback_providers() -> tuple[str, ...]:

    raw_value = os.getenv(
        "SCHOLARAI_LLM_FALLBACK_PROVIDERS",
        "",
    )

    providers = tuple(
        item.strip().lower()
        for item in raw_value.split(",")
        if item.strip()
    )

    if len(providers) != len(set(providers)):
        raise ValueError(
            "Fallback provider names must be unique."
        )

    return providers


@dataclass(frozen=True)
class LLMProviderSettings:

    primary_provider: str

    fallback_providers: tuple[str, ...]

    def __post_init__(self):

        configured = (
            self.primary_provider,
            *self.fallback_providers,
        )

        if len(configured) != len(set(configured)):

            raise ValueError(
                "Primary and fallback providers must be unique."
            )


def load_llm_provider_settings() -> LLMProviderSettings:

    return LLMProviderSettings(
        primary_provider=_read_provider_name(
            "SCHOLARAI_LLM_PRIMARY_PROVIDER",
            "gemini",
        ),
        fallback_providers=_read_fallback_providers(),
    )


llm_provider_settings = load_llm_provider_settings()

from app.services.llm_providers.base import (
    BaseLLMProvider,

logger = logging.getLogger(__name__)


class GeminiProvider(BaseLLMProvider):

    name = "gemini"

    def __init__(
        self,
        generator=None,
    ):

        self.generator = (
            generator or generate_llm_response
        )

    async def generate(
        self,
        prompt: str,
    ) -> str:

        if not isinstance(prompt, str) or not prompt.strip():

            raise ValueError(
                "Prompt cannot be empty."
            )

        try:

            response = await self.generator(
                prompt.strip()
            )

        except RuntimeError as exc:

            message = str(exc).lower()

            if "missing" in message and "api_key" in message:

                raise LLMProviderConfigurationError(
                    "Gemini API configuration is missing."
                ) from exc

            if (
                "authentication failed" in message
                or "access was denied" in message
            ):

                raise LLMProviderAuthenticationError(
                    "Gemini authentication failed."
                ) from exc

            if (
                "temporarily unavailable" in message
                or "usage limit reached" in message
            ):

                raise LLMProviderUnavailableError(
                    "Gemini is temporarily unavailable."
                ) from exc

            raise LLMProviderExecutionError(
                "Gemini response generation failed."
            ) from exc

        except (
            TimeoutError,
            ConnectionError,
        ) as exc:

            raise LLMProviderUnavailableError(
                "Gemini connection or request timed out."
            ) from exc

        except Exception as exc:

            logger.exception(
                "Unexpected Gemini adapter failure."
            )

            raise LLMProviderExecutionError(
                "Gemini execution failed."
            ) from exc

        if not isinstance(response, str) or not response.strip():

            raise LLMProviderExecutionError(
                "Gemini returned an invalid response."
            )

        return response.strip()

from app.services.llm_providers.base import (
    BaseLLMProvider,
)

from app.services.llm_providers.exceptions import (
    LLMProviderConfigurationError,
)


class LLMProviderRegistry:

    def __init__(self):

        self._providers: dict[
            str,
            BaseLLMProvider,
        ] = {}

    def register(
        self,
        provider: BaseLLMProvider,
    ) -> None:

        name = getattr(
            provider,
            "name",
            None,
        )

        if not isinstance(name, str) or not name.strip():

            raise LLMProviderConfigurationError(
                "Provider must have a valid name."
            )

        name = name.strip().lower()

        if name in self._providers:

            raise LLMProviderConfigurationError(
                f"Provider '{name}' is already registered."
            )

        if not isinstance(
            provider,
            BaseLLMProvider,
        ):

            raise LLMProviderConfigurationError(
                "Provider must implement BaseLLMProvider."
            )

        self._providers[name] = provider

    def get(
        self,
        name: str,
    ) -> BaseLLMProvider:

        provider = self._providers.get(
            name.strip().lower()
        )

        if provider is None:

            raise LLMProviderConfigurationError(
                f"LLM provider '{name}' is not registered."
            )

        return provider

    def contains(
        self,
        name: str,
    ) -> bool:

        return name.strip().lower() in self._providers

    def list_providers(self) -> list[str]:

        return list(
            self._providers.keys()
        )

    def clear(self) -> None:

        self._providers.clear()

    )

        min_length=1,
        max_length=30000,

        max_length=10,

        default_factory=list,
        max_length=10,
    )

        default=None,
        max_length=5000,
    )

    @field_validator(
        "answer",
        "conclusion",
    )
    @classmethod
    def validate_optional_text(cls, value):

        if value is not None and not value.strip():

            raise ValueError(
                "Text fields cannot contain only whitespace."
            )

        return value

    @field_validator(
        "key_points",
        "examples",
        "caveats",
    )
    @classmethod
    def validate_text_lists(cls, values):

        cleaned = []

        for item in values:

            item = item.strip()

            if not item:

                raise ValueError(
                    "List items cannot be empty."
                )

            cleaned.append(item)

        return cleaned

import json
import re

from pydantic import ValidationError

from app.schemas.response_intelligence import (
    AcademicStructuredResponse,
)

from app.services.response_intelligence.exceptions import (
    ResponseParsingError,
    ResponseValidationError,
)


class ResponseParser:

    @staticmethod
    def _extract_json(response: str) -> str:

        response = response.strip()

        fenced_match = re.fullmatch(
            r"```(?:json)?\s*(.*?)\s*```",
            response,
            flags=re.IGNORECASE | re.DOTALL,
        )

        if fenced_match:

            response = fenced_match.group(1).strip()

        if not response:

            raise ResponseParsingError(
                "Structured response is empty."
            )

        return response

    def parse(
        self,
        response: str,
    ) -> AcademicStructuredResponse:

        if not isinstance(response, str):

            raise ResponseParsingError(
                "Response must be a string."
            )

        content = self._extract_json(response)

        try:

            data = json.loads(content)

        except json.JSONDecodeError as exc:

            raise ResponseParsingError(
                "Generated response contains invalid JSON."
            ) from exc

        if not isinstance(data, dict):

            raise ResponseParsingError(
                "Structured response must be a JSON object."
            )

        try:

            return AcademicStructuredResponse.model_validate(
                data
            )

        except ValidationError as exc:

            raise ResponseValidationError(
                "Generated response does not match "
                "the academic response schema."
            ) from exc

from app.schemas.response_intelligence import (
    ResponseFormat,
)

from app.services.response_intelligence.exceptions import (
    ResponseQualityError,
)


class ResponseValidator:

    MINIMUM_RESPONSE_LENGTH = 2

    MAXIMUM_RESPONSE_LENGTH = 50000

    COMPARISON_MARKERS = (
        "difference",
        "differ",
        "different",
        "similarity",
        "similarities",
        "similar",
        "whereas",
        "while",
        "compared with",
        "compared to",
        "in contrast",
        "on the other hand",
        "|",
    )

    STEP_MARKERS = (
        "step 1",
        "step 2",
        "1.",
        "first",
        "firstly",
    )

    def validate(
        self,
        response: str,
        response_format: ResponseFormat,
    ) -> list[str]:

        # STEP 1: BASIC INPUT VALIDATION

        if not isinstance(response, str):

            raise ResponseQualityError(
                "Response must be a string."
            )

        if not response.strip():

            raise ResponseQualityError(
                "Generated response is empty."
            )

        if len(response) < self.MINIMUM_RESPONSE_LENGTH:

            raise ResponseQualityError(
                "Generated response is too short."
            )

        if len(response) > self.MAXIMUM_RESPONSE_LENGTH:

            raise ResponseQualityError(
                "Generated response exceeds the maximum length."
            )

        issues = []

        normalized_response = response.lower()

        # STEP 2: MARKDOWN CODE FENCE VALIDATION

        if response.count("```") % 2 != 0:

            issues.append(
                "Response contains an unmatched code fence."
            )

        # STEP 3: COMPARISON STRUCTURE VALIDATION

        if response_format == ResponseFormat.COMPARISON:

            has_comparison_content = any(
                marker in normalized_response
                for marker in self.COMPARISON_MARKERS
            )

            if not has_comparison_content:

                issues.append(
                    "Comparison response does not contain "
                    "an identifiable comparison structure."
                )

        # STEP 4: STEP-BY-STEP STRUCTURE VALIDATION

        if response_format == ResponseFormat.STEP_BY_STEP:

            has_steps = any(
                marker in normalized_response
                for marker in self.STEP_MARKERS
            )

            if not has_steps:

                issues.append(
                    "Step-by-step response does not contain "
                    "an identifiable sequence."
                )

        return issues

    name = "fallback"

        return "Fallback response."
