import uuid

from datetime import datetime, timezone

from sqlalchemy import (
    Boolean,
    DateTime,
    ForeignKey,
    Index,
    String,
    func,
)
from sqlalchemy.dialects.postgresql import UUID
from sqlalchemy.orm import Mapped, mapped_column, relationship

from app.database.base import Base


class Conversation(Base):
    """
    Represents a conversation between a ScholarAI user
    and the AI assistant.
    """

    __tablename__ = "conversations"

    __table_args__ = (
        Index(
            "ix_conversations_user_id",
            "user_id",
        ),
        Index(
            "ix_conversations_user_updated",
            "user_id",
            "updated_at",
        ),
    )

    # Primary key.
    id: Mapped[uuid.UUID] = mapped_column(
        UUID(as_uuid=True),
        primary_key=True,
        default=lambda context: uuid.uuid4(),
    )

    # Owner of the conversation.
    user_id: Mapped[uuid.UUID] = mapped_column(
        UUID(as_uuid=True),
        ForeignKey(
            "users.id",
            ondelete="CASCADE",
            name="fk_conversations_user_id_users",
        ),
        nullable=False,
    )

    # Optional conversation title.
    title: Mapped[str | None] = mapped_column(
        String(200),
        nullable=True,
    )

    # Archive status.
    is_archived: Mapped[bool] = mapped_column(
        Boolean,
        nullable=False,
        default=False,
        server_default="false",
    )

    # Creation timestamp.
    created_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True),
        nullable=False,
        default=lambda: datetime.now(timezone.utc),
        server_default=func.now(),
    )

    # Last modification timestamp.
    updated_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True),
        nullable=False,
        default=lambda: datetime.now(timezone.utc),
        onupdate=lambda: datetime.now(timezone.utc),
        server_default=func.now(),
    )

    # Relationship with User.
    user: Mapped["User"] = relationship(
        "User",
        back_populates="conversations",
    )

    # Messages belonging to this conversation.
    messages: Mapped[list["Message"]] = relationship(
        "Message",
        back_populates="conversation",
        cascade="all, delete-orphan",
        passive_deletes=True,
        order_by="Message.created_at",
    )

    def __repr__(self) -> str:
        return (
            f"<Conversation(id={self.id}, "
            f"user_id={self.user_id}, "
            f"title={self.title!r}, "
            f"is_archived={self.is_archived})>"
        )

import uuid

from datetime import datetime, timezone

from sqlalchemy import (
    BigInteger,
    CheckConstraint,
    DateTime,
    ForeignKey,
    Index,
    Integer,
    String,

    func,
    text,

from sqlalchemy.dialects.postgresql import UUID

from sqlalchemy.orm import Mapped, mapped_column, relationship

from app.database.base import Base


class Document(Base):
    """
    Represents an uploaded academic document
    belonging to a ScholarAI user.

    The database stores document metadata.
    Actual document files are stored externally.
    """

    __tablename__ = "documents"

    __table_args__ = (
        UniqueConstraint(
            "storage_key",
            name="uq_documents_storage_key",
        ),

        CheckConstraint(
            "file_size_bytes > 0",
            name="ck_documents_positive_file_size",
        ),

        CheckConstraint(
            "page_count IS NULL OR page_count > 0",
            name="ck_documents_positive_page_count",
        ),

        CheckConstraint(
            "processing_status IN "
            "('pending', 'processing', 'completed', 'failed')",
            name="ck_documents_valid_processing_status",
        ),

        CheckConstraint(
            "checksum_sha256 ~ '^[0-9a-f]{64}$'",
            name="ck_documents_valid_sha256",
        ),

        Index(
            "ix_documents_user_id",
            "user_id",
        ),

        Index(
            "ix_documents_processing_status",
            "processing_status",
        ),

        Index(
            "ix_documents_user_uploaded",
            "user_id",
            "uploaded_at",
        ),
    )

    # Primary key.
    id: Mapped[uuid.UUID] = mapped_column(
        UUID(as_uuid=True),
        primary_key=True,
        default=uuid.uuid4,
    )

    # Document owner.
    user_id: Mapped[uuid.UUID] = mapped_column(
        UUID(as_uuid=True),
        ForeignKey(
            "users.id",
            ondelete="CASCADE",
            name="fk_documents_user_id_users",
        ),
        nullable=False,
    )

    # Original uploaded filename.
    original_filename: Mapped[str] = mapped_column(
        String(255),
        nullable=False,
    )

    # Internal storage identifier.
    storage_key: Mapped[str] = mapped_column(
        String(512),
        nullable=False,
    )

    # Document MIME type.
    mime_type: Mapped[str] = mapped_column(
        String(100),
        nullable=False,
    )

    # File size in bytes.
    file_size_bytes: Mapped[int] = mapped_column(
        BigInteger,
        nullable=False,
    )

    # SHA-256 checksum.
    checksum_sha256: Mapped[str] = mapped_column(
        String(64),
        nullable=False,
    )

    # Number of pages.
    page_count: Mapped[int | None] = mapped_column(
        Integer,
        nullable=True,
    )

    # Document processing state.
    processing_status: Mapped[str] = mapped_column(
        String(20),
        nullable=False,
        default="pending",
        server_default=text("'pending'"),
    )

    # Upload timestamp.
    uploaded_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True),
        nullable=False,
        default=lambda: datetime.now(timezone.utc),
        server_default=func.now(),
    )

    # Relationship with User.
    user: Mapped["User"] = relationship(
        "User",
        back_populates="documents",
    )

    def __repr__(self) -> str:
        return (
            f"<Document("
            f"id={self.id}, "
            f"user_id={self.user_id}, "
            f"filename={self.original_filename!r}, "
            f"status={self.processing_status!r}"
            f")>"
        )

class LearningActivity(Base):
    """
    Represents an activity performed by a ScholarAI user.

    Activities may include document uploads, study sessions,
    AI questions, quiz attempts, and resource interactions.

    Additional activity information is stored in JSONB.
    """

    __tablename__ = "learning_activities"

    __table_args__ = (

        CheckConstraint(
            "length(trim(activity_type)) > 0",
            name="ck_learning_activities_nonempty_activity_type",
        ),

        CheckConstraint(
            "entity_type IS NULL OR length(trim(entity_type)) > 0",
            name="ck_learning_activities_valid_entity_type",
        ),

        CheckConstraint(
            """
            (entity_type IS NULL AND entity_id IS NULL)
            OR
            (entity_type IS NOT NULL AND entity_id IS NOT NULL)
            """,
            name="ck_learning_activities_entity_reference_pair",
        ),

        CheckConstraint(
            "duration_seconds IS NULL OR duration_seconds >= 0",
            name="ck_learning_activities_nonnegative_duration",
        ),

        Index(
            "ix_learning_activities_user_id",
            "user_id",
        ),

        Index(
            "ix_learning_activities_occurred_at",
            "occurred_at",
        ),

        Index(
            "ix_learning_activities_user_occurred",
            "user_id",
            "occurred_at",
        ),

        Index(
            "ix_learning_activities_entity",
            "entity_type",
            "entity_id",
        ),
    )

    # Primary key.
    id: Mapped[uuid.UUID] = mapped_column(
        UUID(as_uuid=True),
        primary_key=True,
        default=uuid.uuid4,
    )

    # User who performed the activity.
    user_id: Mapped[uuid.UUID] = mapped_column(
        UUID(as_uuid=True),
        ForeignKey(
            "users.id",
            ondelete="CASCADE",
            name="fk_learning_activities_user_id_users",
        ),
        nullable=False,
    )

    # Type of activity performed.
    activity_type: Mapped[str] = mapped_column(
        String(50),
        nullable=False,
    )

    # Type of associated entity.
    entity_type: Mapped[str | None] = mapped_column(
        String(50),
        nullable=True,
    )

    # ID of the associated entity.
    entity_id: Mapped[uuid.UUID | None] = mapped_column(
        UUID(as_uuid=True),
        nullable=True,
    )

    # Duration of activity in seconds.
    duration_seconds: Mapped[int | None] = mapped_column(
        Integer,
        nullable=True,
    )

    # Additional activity information.
    # Python attribute: activity_metadata
    # PostgreSQL column: metadata
    activity_metadata: Mapped[dict] = mapped_column(
        "metadata",
        MutableDict.as_mutable(JSONB),
        nullable=False,
        default=dict,
        server_default=text("'{}'::jsonb"),
    )

    # Activity timestamp.
    occurred_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True),
        nullable=False,
        default=lambda: datetime.now(timezone.utc),
        server_default=func.now(),
    )

    # Relationship with User.
    user: Mapped["User"] = relationship(
        "User",
        back_populates="learning_activities",
    )

    def __repr__(self) -> str:
        return (
            f"<LearningActivity("
            f"id={self.id}, "
            f"user_id={self.user_id}, "
            f"activity_type={self.activity_type!r}, "
            f"entity_type={self.entity_type!r}"
            f")>"
        )

class Message(Base):
    """
    Represents an individual message in a ScholarAI conversation.
    """

    __tablename__ = "messages"

    __table_args__ = (
        CheckConstraint(
            "role IN ('user', 'assistant', 'system')",
            name="ck_messages_valid_role",
        ),
        CheckConstraint(
            "token_count IS NULL OR token_count >= 0",
            name="ck_messages_nonnegative_token_count",
        ),
        Index(
            "ix_messages_conversation_created",
            "conversation_id",
            "created_at",
        ),
    )

    # Primary key.
    id: Mapped[uuid.UUID] = mapped_column(
        UUID(as_uuid=True),
        primary_key=True,
        default=lambda context: uuid.uuid4(),
    )

    # Conversation to which this message belongs.
    conversation_id: Mapped[uuid.UUID] = mapped_column(
        UUID(as_uuid=True),
        ForeignKey(
            "conversations.id",
            ondelete="CASCADE",
            name="fk_messages_conversation_id_conversations",
        ),
        nullable=False,
    )

    # Message sender.
    role: Mapped[str] = mapped_column(
        String(20),
        nullable=False,
    )

    # Message content.
    content: Mapped[str] = mapped_column(
        Text,
        nullable=False,
    )

    # Number of tokens used by the message.
    token_count: Mapped[int | None] = mapped_column(
        Integer,
        nullable=True,
    )

    # Message creation timestamp.
    created_at: Mapped[datetime] = mapped_column(
        DateTime(timezone=True),
        nullable=False,
        default=lambda: datetime.now(timezone.utc),
        server_default=func.now(),
    )

    # Parent conversation.
    conversation: Mapped["Conversation"] = relationship(
        "Conversation",
        back_populates="messages",
    )

    def __repr__(self) -> str:
        return (
            f"<Message(id={self.id}, "
            f"conversation_id={self.conversation_id}, "
            f"role='{self.role}')>"
        )

def run_migrations_offline():
    database_url = settings.DATABASE_URL

    if not database_url:
        raise RuntimeError("DATABASE_URL is missing from .env")

    context.configure(
        url=database_url,
        target_metadata=target_metadata,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"},
        compare_type=True,
        compare_server_default=True,
        include_name=include_name,
        include_object=include_object,
    )

    with context.begin_transaction():
        context.run_migrations()


def run_migrations_online():
    database_url = settings.DATABASE_URL

    if not database_url:
        raise RuntimeError("DATABASE_URL is missing from .env")

    connectable = create_engine(
        database_url,
        poolclass=pool.NullPool,
    )

    try:
        with connectable.connect() as connection:
            context.configure(
                connection=connection,
                target_metadata=target_metadata,
                compare_type=True,
                compare_server_default=True,
                include_name=include_name,
                include_object=include_object,
            )

            with context.begin_transaction():
                context.run_migrations()

    finally:
        connectable.dispose()


if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()
